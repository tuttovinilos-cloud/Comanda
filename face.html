create table if not exists personas ( id uuid primary key default gen_random_uuid(), nombre text not null, cargo text, activo boolean default true, created_at timestamp with time zone default now() ); insert into personas (nombre, cargo) select 'Roberto', 'CEO' where not exists (select 1 from personas where nombre = 'Roberto'); insert into personas (nombre, cargo) select 'Rubén', 'Operador impresión' where not exists (select 1 from personas where nombre = 'Rubén'); insert into personas (nombre, cargo) select 'Ricardo', 'Atención al cliente' where not exists (select 1 from personas where nombre = 'Ricardo'); insert into personas (nombre, cargo) select 'Chico', 'Producción' where not exists (select 1 from personas where nombre = 'Chico'); insert into personas (nombre, cargo) select 'Carlos', 'Ayudante' where not exists (select 1 from personas where nombre = 'Carlos'); create table if not exists rostros_referencia ( id uuid primary key default gen_random_uuid(), persona_id uuid references personas(id) on delete cascade, persona_nombre text, foto_url text, descriptor jsonb not null, distancia_base numeric, created_at timestamp with time zone default now() ); create table if not exists fotos_registro ( id uuid primary key default gen_random_uuid(), fecha timestamp with time zone default now(), imagen_url text, persona_id uuid references personas(id), persona_detectada text, confianza numeric, distancia numeric, tipo_registro text, nota text, created_at timestamp with time zone default now() ); alter table personas enable row level security; alter table rostros_referencia enable row level security; alter table fotos_registro enable row level security; drop policy if exists "leer personas" on personas; create policy "leer personas" on personas for select to anon using (true); drop policy if exists "leer rostros referencia" on rostros_referencia; create policy "leer rostros referencia" on rostros_referencia for select to anon using (true); drop policy if exists "insertar rostros referencia" on rostros_referencia; create policy "insertar rostros referencia" on rostros_referencia for insert to anon with check (true); drop policy if exists "leer fotos registro" on fotos_registro; create policy "leer fotos registro" on fotos_registro for select to anon using (true); drop policy if exists "insertar fotos registro" on fotos_registro; create policy "insertar fotos registro" on fotos_registro for insert to anon with check (true); drop policy if exists "subir fotos publicas" on storage.objects; create policy "subir fotos publicas" on storage.objects for insert to anon with check (bucket_id = 'fotos-registro'); drop policy if exists "leer fotos publicas" on storage.objects; create policy "leer fotos publicas" on storage.objects for select to anon using (bucket_id = 'fotos-registro');
