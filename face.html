Copia esto en Supabase → SQL Editor. Si ya lo hiciste, ignora esta parte.

create table if not exists personas ( id uuid primary key default gen_random_uuid(), nombre text not null, cargo text, activo boolean default true, created_at timestamp with time zone default now() ); insert into personas (nombre, cargo) select 'Roberto', 'CEO' where not exists (select 1 from personas where nombre = 'Roberto'); insert into personas (nombre, cargo) select 'Rubén', 'Operador impresión' where not exists (select 1 from personas where nombre = 'Rubén'); insert into personas (nombre, cargo) select 'Ricardo', 'Atención al cliente' where not exists (select 1 from personas where nombre = 'Ricardo'); insert into personas (nombre, cargo) select 'Chico', 'Producción' where not exists (select 1 from personas where nombre = 'Chico'); insert into personas (nombre, cargo) select 'Carlos', 'Ayudante' where not exists (select 1 from personas where nombre = 'Carlos'); create table if not exists fotos_registro ( id uuid primary key default gen_random_uuid(), fecha timestamp with time zone default now(), imagen_url text, persona_id uuid references personas(id), persona_detectada text, confianza numeric, tipo_registro text, nota text, created_at timestamp with time zone default now() ); alter table personas enable row level security; alter table fotos_registro enable row level security; drop policy if exists "leer personas" on personas; create policy "leer personas" on personas for select using (true); drop policy if exists "leer fotos registro" on fotos_registro; create policy "leer fotos registro" on fotos_registro for select using (true); drop policy if exists "insertar fotos registro" on fotos_registro; create policy "insertar fotos registro" on fotos_registro for insert with check (true); drop policy if exists "subir fotos publicas" on storage.objects; create policy "subir fotos publicas" on storage.objects for insert with check (bucket_id = 'fotos-registro'); drop policy if exists "leer fotos publicas" on storage.objects; create policy "leer fotos publicas" on storage.objects for select using (bucket_id = 'fotos-registro');
